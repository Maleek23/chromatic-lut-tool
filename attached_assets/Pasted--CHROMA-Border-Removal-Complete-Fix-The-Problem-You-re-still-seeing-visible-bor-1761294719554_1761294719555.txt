## **Visual Debugging Script:**

```javascript
// Run this in browser console to find borders
document.querySelectorAll('*').forEach(el => {
  const styles = window.getComputedStyle(el);
  
  // Check for borders
  if (styles.borderTopWidth !== '0px' ||
      styles.borderBottomWidth !== '0px' ||
      styles.borderLeftWidth !== '0px' ||
      styles.borderRightWidth !== '0px') {
    console.log('Border found:', el);
    el.style.outline = '2px solid red'; // Highlight it
  }
  
  // Check for box-shadows that look like borders
  if (styles.boxShadow !== 'none' && 
      styles.boxShadow.includes('inset')) {
    console.log('Inset shadow (looks like border):', el);
    el.style.outline = '2px solid orange';
  }
});
```
